{{- /* Reads an event's startDate/startTime/endDate/endTime (all Central time). Times may be written "6pm", "6:30 PM" or "18:30"; returns them as "6:30 PM". Returns: start, end (dates), startTime, endTime ("" when not given). */ -}} {{- $start := .Date -}} {{- $end := (.Params.endDate | default $start) | time -}} {{- $times := dict -}} {{- range $key := slice "startTime" "endTime" -}} {{- $out := "" -}} {{- with index $.Params $key -}} {{- $s := string . -}} {{- $m := findRESubmatch `(?i)^\s*(\d{1,2})(?::([0-5]\d))?\s*(?:([ap])\.?\s*m?\.?)?\s*$` $s 1 -}} {{- if not $m -}} {{- errorf "%s: %s %q isn't a time Hugo understands. Write it like \"6:00 PM\"." $.File.Path $key $s -}} {{- end -}} {{- $g := index $m 0 -}} {{- $h := int (strings.TrimLeft "0" (index $g 1) | default "0") -}} {{- $min := index $g 2 | default "00" -}} {{- $ap := lower (index $g 3) -}} {{- if or (gt $h 23) (and $ap (or (lt $h 1) (gt $h 12))) -}} {{- errorf "%s: %s %q isn't a real time. Write it like \"6:00 PM\"." $.File.Path $key $s -}} {{- end -}} {{- if and (eq $ap "p") (lt $h 12) }}{{ $h = add $h 12 }}{{ end -}} {{- if and (eq $ap "a") (eq $h 12) }}{{ $h = 0 }}{{ end -}} {{- $out = (time (printf "2000-01-01T%02d:%s:00" $h $min)).Format "3:04 PM" -}} {{- end -}} {{- $times = merge $times (dict $key $out) -}} {{- end -}} {{- return dict "start" $start "end" $end "startTime" $times.startTime "endTime" $times.endTime -}}
