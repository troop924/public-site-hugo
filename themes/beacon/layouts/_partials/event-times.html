{{- /* Reads an event's startDate/startTime/endDate/endTime (all Central time). Returns: start, end (dates), startTime, endTime ("6:30 PM" or ""), startClock ("18:30" or "", for sorting). */ -}} {{- $start := .Date -}} {{- $end := (.Params.endDate | default $start) | time -}} {{- $s := partial "parse-time.html" (dict "value" .Params.startTime "where" (printf "%s: startTime" .File.Path)) -}} {{- $e := partial "parse-time.html" (dict "value" .Params.endTime "where" (printf "%s: endTime" .File.Path)) -}} {{- return dict "start" $start "end" $end "startTime" $s.label "endTime" $e.label "startClock" $s.clock -}}
